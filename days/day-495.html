المستخدم بيكتب: "what does our team do when uploads fail and the budget is gone?". وفي الـ corpus فيه مستند بيقول: "AbortMultipartOnFail aborts an in-flight S3 multipart upload and decrements the per-bucket retry budget when the upload fails".

السؤال والمستند مفيش بينهم ولا noun phrase مشتركة. الـ BM25 بيفوّته. والـ bi-encoder بيحطه تالت أو رابع، لأن vector السؤال وقع في منطقة من الـ embedding space قريبة من مستند عن الـ "cancelled jobs" مش عن الـ "aborted uploads". الـ reranker بتاع امبارح ممكن ينقذ الموقف لو المستند وصل الـ top-N، بس لو موصلش، الـ reranker عمره ما هيشوفه.

    Multi-Query وDecomposition

    Multi-Query expansion

    ولّد N صيغة (paraphrase) للسؤال، هات top-k بكل صيغة، وادمج الـ N قوايم بالـ RRF (نفس الخوارزمية من درس 65). رخيص، بيشتغل parallel، وdeterministic.

    Rewrite the following question in {N} different ways. Each rewrite must preserve
the original intent. Number them 1 to {N}. Do not add explanations.
    • بيكسب لما صياغة المستخدم واحدة من صيغ كتير صح، وأي صيغة تانية كانت هتسأل أحسن.
    • بيخسر لما كل الصيغ وحشة بنفس الطريقة، لأن الأصل نفسه كان وحش بالشكل ده.
    تشبيه: زي ما تسأل ٣ أصحاب يدوّروا على نفس الحاجة في جوجل، كل واحد بكلامه. اللي يظهر عند أكتر من واحد فيهم غالبًا هو الصح. ده بالظبط اللي الـ RRF بيكافئه.

      خطوة ١: tokenizer و embedding وهمي deterministic

      from __future__ import annotations
import math, re
from collections import Counter, defaultdict
from dataclasses import dataclass, field

_TOKEN = re.compile(r"[a-z0-9]+")

def tokenize(text: str) -> list[str]:
    return _TOKEN.findall(text.lower())

def mock_embed(text: str, dim: int = 96) -> list[float]:
    vec = [0.0] * dim
    for tok in tokenize(text):
        h = 0
        for ch in tok:                         # hash ثابت لكل كلمة
            h = (h * 1315423911) ^ ord(ch)
            h &= 0xFFFFFFFF
        vec[h % dim] += 1.0
        vec[(h >> 7) % dim] += 0.5
        for i in range(len(tok) - 1):          # bigrams حروف: كلمات شبه بعض تقرب
            bg = (ord(tok[i]) * 31 + ord(tok[i + 1])) & 0xFFFFFFFF
            vec[bg % dim] += 0.25
    norm = math.sqrt(sum(v * v for v in vec)) or 1.0
    return [v / norm for v in vec]             # normalized → الـ dot = cosine

def cosine(a, b):
    return sum(x * y for x, y in zip(a, b))

      خطوة ٢: الـ retriever (مبسط من درس 65)

      @dataclass
class Doc:
    doc_id: str
    title: str
    body: str
    def field_text(self) -> str:
        return f"{self.title}\n{self.body}"

@dataclass
class BM25Index:
    k1: float = 1.5
    b: float = 0.75
    docs: list = field(default_factory=list)
    doc_lens: list = field(default_factory=list)
    df: Counter = field(default_factory=Counter)
    tf: list = field(default_factory=list)
    avgdl: float = 0.0

    def add(self, doc: Doc) -> None:
        toks = tokenize(doc.title) * 3 + tokenize(doc.body)   # العنوان وزنه ×3
        counts = Counter(toks)
        self.docs.append(doc); self.doc_lens.append(len(toks)); self.tf.append(counts)
        for term in counts:
            self.df[term] += 1
        self.avgdl = sum(self.doc_lens) / max(1, len(self.doc_lens))

    def search(self, query: str, k: int):
        n = len(self.docs)
        if n == 0:
            return []
        scores = [0.0] * n
        for term in tokenize(query):
            df = self.df.get(term, 0)
            if df == 0:
                continue
            idf = math.log((n - df + 0.5) / (df + 0.5) + 1.0)
            for i, counts in enumerate(self.tf):
                f = counts.get(term, 0)
                if f == 0:
                    continue
                dl = self.doc_lens[i]
                denom = f + self.k1 * (1 - self.b + self.b * dl / (self.avgdl or 1))
                scores[i] += idf * f * (self.k1 + 1) / denom
        return sorted(zip(self.docs, scores), key=lambda x: -x[1])[:k]

@dataclass
class DenseIndex:
    vectors: list = field(default_factory=list)
    def add(self, doc):
        self.vectors.append((doc, mock_embed(doc.field_text())))
    def search_vec(self, qv, k):
        scored = [(d, cosine(qv, v)) for d, v in self.vectors]
        scored.sort(key=lambda x: -x[1])
        return scored[:k]
    def search(self, query, k):
        return self.search_vec(mock_embed(query), k)

def rrf(rankings, k: int = 60):
    score = defaultdict(float); by_id = {}
    for ranks in rankings:
        for rank, (doc, _) in enumerate(ranks):
            score[doc.doc_id] += 1.0 / (k + rank + 1)   # بالترتيب مش بالـ score
            by_id[doc.doc_id] = doc
    fused = sorted(score.items(), key=lambda x: -x[1])
    return [(by_id[did], s) for did, s in fused]

@dataclass
class HybridRetriever:
    bm25: BM25Index = field(default_factory=BM25Index)
    dense: DenseIndex = field(default_factory=DenseIndex)
    def add(self, doc):
        self.bm25.add(doc); self.dense.add(doc)
    def search(self, query, k_each=5, k_out=5):
        return rrf([self.bm25.search(query, k_each), self.dense.search(query, k_each)])[:k_out]
    def search_vec(self, qv, qtext, k_each=5, k_out=5):
        # حالة HyDE: الـ dense بالـ vector المزيف، والـ BM25 على السؤال الأصلي
        b = self.bm25.search(qtext, k_each)
        d = self.dense.search_vec(qv, k_each)
        return rrf([b, d])[:k_out]

      خطوة ٣: الـ MockLLM

      انسخ الجداول _SYNONYMS وHYDE_TABLE وMQ_TABLE وDECOMP_TABLE من الملف الأصلي (دي داتا مش منطق). المنطق هو ده:

      @dataclass
class MockLLM:
    def generate_hypothetical(self, query: str) -> str:
        key = query.lower().strip().rstrip("?").strip()
        if key in HYDE_TABLE:
            return HYDE_TABLE[key]
        toks = tokenize(query)                       # fallback: وسّع بالمرادفات
        expanded = []
        for t in toks:
            expanded.append(t)
            expanded.extend(_SYNONYMS.get(t, []))
        return " ".join(expanded)

    def paraphrase(self, query: str, n: int = 3) -> list[str]:
        key = query.lower().strip().rstrip("?").strip()
        if key in MQ_TABLE:
            return MQ_TABLE[key][:n]
        toks = tokenize(query); out = []
        for shift in range(n):                       # fallback: بدّل مرادفات بالتناوب
            swapped = []
            for i, t in enumerate(toks):
                opts = _SYNONYMS.get(t, [])
                if opts and (i + shift) % 2 == 0:
                    swapped.append(opts[shift % len(opts)])
                else:
                    swapped.append(t)
            out.append(" ".join(swapped))
        return out

    def decompose(self, query: str) -> list[str]:
        key = query.lower().strip().rstrip("?").strip()
        if key in DECOMP_TABLE:
            return DECOMP_TABLE[key]
        if " and " in query.lower():                 # fallback: اقسم على and
            parts = re.split(r"\s+and\s+", query, flags=re.IGNORECASE)
            return [p.strip().rstrip("?") for p in parts if p.strip()]
        return [query]                               # atomic → رجّعه زي ما هو

      خطوة ٤: الـ rewriters (interface صغير)

      @dataclass
class RewriteResult:
    strategy: str
    rewrites: list[str]
    hypothetical: str | None = None

class Rewriter:
    name: str
    def rewrite(self, query: str) -> RewriteResult:
        raise NotImplementedError

@dataclass
class HyDERewriter(Rewriter):
    llm: MockLLM = field(default_factory=MockLLM)
    name: str = "hyde"
    def rewrite(self, query):
        h = self.llm.generate_hypothetical(query)
        return RewriteResult(self.name, rewrites=[query], hypothetical=h)

@dataclass
class MultiQueryRewriter(Rewriter):
    llm: MockLLM = field(default_factory=MockLLM)
    n: int = 3
    name: str = "multiquery"
    def rewrite(self, query):
        return RewriteResult(self.name, rewrites=[query] + self.llm.paraphrase(query, n=self.n))

@dataclass
class DecomposeRewriter(Rewriter):
    llm: MockLLM = field(default_factory=MockLLM)
    name: str = "decompose"
    def rewrite(self, query):
        return RewriteResult(self.name, rewrites=self.llm.decompose(query))

class _IdentityRewriter(Rewriter):                  # الـ baseline: من غير rewrite
    name = "no-rewrite"
    def rewrite(self, query):
        return RewriteResult(self.name, rewrites=[query])

      خطوة ٥: الـ retrieval من خلال الـ rewriter

      def retrieve_with_rewriter(query, rewriter, retriever, k_each=5, k_out=5):
    rw = rewriter.rewrite(query)
    rankings = []
    if rw.hypothetical is not None:                  # HyDE: ranking إضافية بالـ vector المزيف
        hv = mock_embed(rw.hypothetical)
        rankings.append(retriever.search_vec(hv, qtext=query, k_each=k_each, k_out=k_each))
    for r in rw.rewrites:                            # ranking لكل صيغة / sub-question
        rankings.append(retriever.search(r, k_each=k_each, k_out=k_each))
    fused = rrf(rankings)[:k_out]                    # دمج الكل بالـ RRF
    return {"rewriter": rw.strategy, "rewrites": rw.rewrites,
            "hypothetical": rw.hypothetical, "results": fused}

      لاحظ: في HyDE الـ rewrites=[query]، يعني بيتعمل ranking بالسؤال الأصلي كمان جنب ranking الـ hypothetical، والاتنين بيتدمجوا.

            marker = "  <- expected winner" if name == expected_winner else ""
            print(f"{name:<12} | {q[:58]:<60} |  {hit}   |   {gold_rank}{marker}")
        print()

if __name__ == "__main__":
    main()
      شكل الناتج

      جدول فيه ١٢ سطر (٣ أسئلة × ٤ استراتيجيات) بالأعمدة strategy | query | gold@1? | gold-rank، وعلامة <- expected winner جنب الاستراتيجية المتصممة للسؤال ده. جدول المقارنة تحت معمل ٢ بيحسب نفس الأرقام بنفس الخوارزمية في المتصفح، فقارن الرقمين: لو اختلفوا، دوّر على الفرق في الكود بتاعك.

      الـ tests لازم تطلع OK. أهمها: الـ decompose لسؤال atomic بيرجّعه زي ما هو، وHyDE بيغير الترتيب فعلًا، ولكل سؤال GOLD فيه rewriter واحد على الأقل مش أسوأ من الـ baseline.

      05

      الـ failure modes اللي الـ demo مخبيها

      1. HyDE بيألّف identifiers غلط. الموديل بيخترع اسم function. الاسم المتألف بيبقى token وزنه عالي في الـ BM25 ومش موجود في الـ index، فالـ BM25 score للمستند الصح بينهار. الحل: حدّد طول الـ hypothetical، وقلل وزن الـ BM25 في الـ fusion.
      2. صيغ الـ Multi-Query كلها شبه بعض. موديل ضعيف بيطلع ٣ paraphrases شبه متطابقة، فالـ N retrievals بيرجعوا نفس الـ top-k، والـ RRF مش أحسن من retrieval واحد. الحل: تعليمة صريحة للتنوع في الـ prompt، واكتشف التكرار بالـ Jaccard.
      3. الـ Decomposition بيقسّم زيادة (over-split). سؤال atomic بيتحول لقايمة، والـ retrievals كلها بترجع نفس المستند بس بترتيب أقل، فالدمج بيبقى أسوأ من الأصل. الحل: خطوة «هل الأسئلة الفرعية دي مختلفة كفاية؟» قبل الـ fan-out.
      4. الـ latency بتتضاعف. HyDE = LLM call واحد. Multi-Query = LLM call واحد يولّد N صيغة، وبعدين N retrievals. Decomposition = LLM call واحد، وبعدين M retrievals. الـ retrievals بتشتغل parallel، والـ LLM call هو الأرضية (floor).

      معمل ٣: كاشف التكرار وحاسبة الـ latency

      فوق: اكتب ٣ صيغ (أو sub-questions). المعمل بيحسب الـ Jaccard بين كل اتنين (عدد الكلمات المشتركة ÷ عدد الكلمات الكلية). لو أي زوج فوق الـ threshold، الصيغ دي مش هتضيف تنوع، أو الـ decomposition عمل over-split.

      تحت: حرّك زمن الـ LLM call وزمن الـ retrieval وعدد الصيغ، وبدّل بين parallel وserial، وشوف الأرضية بتيجي منين.

      06

      Use It و Ship It: إزاي بيتستخدم في الـ production

      • اختيار الاستراتيجية حسب السؤال: سؤال قصير atomic → Multi-Query. سؤال معقد فيه كذا clause → Decomposition. سؤال مليان jargon → HyDE.
      • Cache بالـ hash بتاع السؤال: أسئلة كتير بتتكرر، فخزّن ناتج الـ rewriter.
      • شغّل التلاتة parallel وادمجهم بالـ RRF: التكلفة ٣ LLM calls ودمج واحد، والجودة = اتحاد تغطية التلاتة.
      • ليه RRF مش score interpolation؟ نفس كلام درس 65: الـ interpolation محتاج calibration لكل corpus (ضبط alpha). الـ RRF بيشتغل على الترتيب، فمش محتاج calibration وبيفضل ثابت مهما كانت مخرجات الـ rewriters.

      Ship It: درس 69 بيركّب مرحلة الـ rewriter دي قبل الـ retriever بتاع درس 65 والـ reranker بتاع درس 66. ودرس 68 (بكرة) بيقيس قد إيه الـ rewriter رفع الـ recall.

      query → rewriter (L67) → hybrid retriever (L65) → cross-encoder rerank (L66) → answer · measured by L68
      07

      التمارين

      1. RAG-Fusion: نفّذ النسخة دي من الـ multi-query (2024) اللي فيها الـ paraphrases متنوعة عن قصد، وبعدين خطوة الـ rerank (درس 66) هي اللي تختار القايمة النهائية.
        تلميحعدّل الـ prompt (أو الـ mock) يطلب صيغ من زوايا مختلفة (سؤال عن السبب، عن الإجراء، عن الاسم التقني). اعمل retrieval لكل صيغة، ادمج بالـ RRF لـ top-N (مثلًا 8)، وبعدين ابعت الـ N دول للـ rerank(query, candidates, top_k) بتاع امبارح بالسؤال الأصلي. استخدم Jaccard (معمل ٣) عشان تتأكد إن الصيغ فعلًا متنوعة.
      2. Step-back prompting: ضيف استراتيجية رابعة: اسأل الـ LLM عن السؤال الأعم، اعمل retrieval بيه، وبعدين ضيّق. قارن على الـ fixture.
        تلميحاعمل StepBackRewriter بيرجّع rewrites=[general_q, query]. في الـ mock ضيف جدول STEPBACK_TABLE، مثلًا "what do we do when a transfer breaks halfway" → "how does the storage service handle upload failures". «ضيّق» هنا معناها إن السؤال الأصلي بيفضل في الدمج، أو تعمل rerank بالسؤال الأصلي. ضيفه للـ dict بتاع strategies في main() وقارن الـ gold-rank.
      3. كاشف الـ atomic: درّب الـ decomposer يعرف الأسئلة الـ atomic بإضافة head «هل السؤال atomic؟». قيس الـ over-split rate قبل وبعد.
        تلميحابدأ بـ heuristic قبل التدريب: مفيش and / vs / compare، أو عدد الكلمات المهمة قليل → atomic. اعمل مجموعة صغيرة أسئلة متعلّمة (atomic / multi). الـ over-split rate = عدد الأسئلة الـ atomic اللي اتقسمت ÷ عدد الأسئلة الـ atomic. لاحظ إن الـ fallback الحالي بيقسم على أي "and"، يعني "search and replace" هتتقسم غلط. ده مثال كويس تبدأ بيه.
      4. Model حقيقي: بدّل الـ MockLLM بـ model call حقيقي وقيس الـ latency لكل استراتيجية على الـ stack بتاعك.
        تلميحاعمل class فيه نفس الـ ٣ methods (generate_hypothetical, paraphrase, decompose) بالـ prompts اللي في محطة ٢ و٣، ومرّره كـ llm=. الـ retriever مش هيتغير. قيس بـ time.perf_counter() حوالين الـ LLM call لوحده وحوالين الـ retrievals لوحدها. هتلاقي الـ LLM call مسيطر على الوقت زي ما معمل ٣ بيقول. وparse الأرقام في رد الـ paraphrase بحذر ("1. ...").
      5. Confidence لكل rewrite: ضيف confidence score لكل صيغة، واحذف اللي تحت الـ threshold. قيس تأثيره على الـ recall.
        تلميحمصدر بسيط للـ confidence: الـ cosine بين mock_embed(rewrite) وmock_embed(query). لو قليل أوي يبقى الصيغة بعدت عن المعنى. لو عالي أوي (قريب من 1) يبقى نسخة مكررة. خلي الـ threshold بين الاتنين، وقيس recall@k على الـ GOLD قبل وبعد. بكرة هتتعلم المقاييس دي بالظبط.
      6. إضافة مني · Router + Cache زي الشغل الحقيقي: اكتب route(query) -> Rewriter: لو فيه " and " أو أكتر من 12 كلمة → decompose، لو فيه token شكله identifier (فيه _ أو CamelCase) → hyde، غير كده → multiquery. وغلّفه بـ cache dict مفتاحه hashlib.sha256(query.lower().strip().encode()).hexdigest(). شغّله على أسئلة الـ GOLD واطبع كام مرة الـ cache اتلقط لما تكرر نفس الأسئلة.
        الإجابةقواعد الـ routing دي تطبيق مباشر لفقرة «Use It» (الطول/التعقيد/الـ jargon). الـ cache لازم يخزّن ناتج الـ rewriter (RewriteResult) مش النتايج النهائية، عشان لو الـ index اتحدّث، الـ retrieval يتعمل من جديد بس من غير ما تدفع LLM call تاني. في الشغل الحقيقي حط TTL للـ cache وطبّع السؤال (lowercase, strip, شيل علامة الاستفهام) زي ما الـ mock بيعمل في الـ key.
      08

      إضافة مني: الحدس، الأخطاء الشائعة، وفين بيظهر في الـ LLMs والـ Agents

      • الحدس في جملة: الأسئلة والإجابات عايشين في «أحياء» مختلفة في الـ embedding space. الأسئلة قصيرة وفيها كلمات استفهام، والمستندات طويلة وفيها مصطلحات. HyDE بيحوّل السؤال لـ «ساكن» في حي الإجابات.
      • متعرضش الـ hypothetical للمستخدم: ده نص ممكن يكون غلط تمامًا. استخدامه الوحيد هو الـ retrieval. لو ظهر في الإجابة النهائية أو في الـ context بتاع الـ generation، هتدخّل hallucination بإيدك.
      • متنساش السؤال الأصلي: الكود بيحط السؤال الأصلي في rewrites في HyDE وفي Multi-Query. لو شلته، ممكن rewrite وحش يضيّع حالة كانت الـ baseline هتجيبها صح.
      • متقيسش بالإحساس: «HyDE حلو» مش قرار. قيس الـ recall@k قبل وبعد على eval set من أسئلتك الحقيقية (ده درس بكرة). الـ fixture هنا متصمم عشان كل استراتيجية تكسب مرة، والداتا بتاعتك مش كده.
      • في الـ Agents: الـ agent اللي بيعمل tool call لـ search بيعمل query rewriting ضمنيًا: الـ LLM نفسه بيكتب الـ search query من المحادثة. ونفس فكرة الـ decomposition هي الأساس بتاع الـ multi-hop / deep research agents: كسّر السؤال، دوّر لكل جزء، ادمج.
      • في المحادثات: سؤال زي «طب وفي الحالة التانية؟» مالوش معنى لوحده. في RAG الشات بنعمل rewrite بالـ history الأول عشان يبقى سؤال مستقل (standalone)، وده بيحصل قبل أي استراتيجية من التلاتة.
